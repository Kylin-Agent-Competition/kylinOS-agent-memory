```rust
use std::env;
use std::process;

fn main() {
    let args: Vec<String> = env::args().collect();

    let mut name = String::from("世界");
    let mut count = 1u32;

    let mut i = 1;
    while i < args.len() {
        match args[i].as_str() {
            "-h" | "--help" => {
                println!("用法: {} [--name <名字>] [--count <次数>]", args[0]);
                println!("  -h, --help       显示帮助");
                println!("  --name <名字>    要问候的名字（默认：世界）");
                println!("  --count <次数>   问候次数（默认：1）");
                return;
            }
            "--name" => {
                i += 1;
                if i >= args.len() {
                    eprintln!("错误：--name 需要一个值");
                    process::exit(1);
                }
                name = args[i].clone();
            }
            "--count" => {
                i += 1;
                if i >= args.len() {
                    eprintln!("错误：--count 需要一个值");
                    process::exit(1);
                }
                count = args[i].parse().unwrap_or_else(|_| {
                    eprintln!("错误：--count 的值必须是数字");
                    process::exit(1);
                });
            }
            other => {
                eprintln!("错误：无法识别的参数：{}", other);
                eprintln!("使用 --help 查看用法");
                process::exit(1);
            }
        }
        i += 1;
    }

    for _ in 0..count {
        println!("你好，{}！", name);
    }
}
```

## 支持的功能

- `--name <名字>`：指定要问候的名字，默认是“世界”。
- `--count <次数>`：指定问候次数，默认是 1。
- `-h` / `--help`：显示帮助信息。
- 对缺失值、非法数字、未知参数进行简单错误处理。
